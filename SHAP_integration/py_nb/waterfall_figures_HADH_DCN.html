def find_gene_index(rna,gene_name, rank=0):
    trans = rna.loc[rna["gene_name"].eq(gene_name),"shuffle_i"]
    print("Transcripts: {}".format(len(trans)))
    gene_index = trans.iloc[rank]
    print("Real logFC / Predicted logFC: \n {}".format(rna.loc[rna["shuffle_i"] == gene_index][["logratio", "pred_logfc"]]))
    return(gene_index)

def simple_waterfall( shap_values, rna,gene_name, rank=0):
    gene_index = find_gene_index(rna, gene_name, rank=rank)
    shap.waterfall_plot(shap_values[gene_index], show=False)
    plt.title("{}; logfc rank {}".format(gene_name, rank))
    plt.tight_layout()
    filename = "../output/figs/{}_{}_waterfall.pdf".format(gene_name, rank)
    plt.savefig(filename)
    plt.show()
    plt.close()

def simple_force( shap_values, rna,gene_name, rank=0):
    gene_index = find_gene_index(rna, gene_name, rank=rank)
    shap.plots.force(shap_values.base_values[gene_index],shap_values.values[gene_index], show=False, matplotlib=True, 
                 feature_names=["atac_logfc","H3K27ac","H3K27me3", "H3K36me3" ,'H3K4me1' ,'H3K4me3','H3K9me3'])
    plt.title("{}; logfc rank {}".format(gene_name, rank), y=1.5)
    plt.tight_layout()
    filename = "../output/figs/{}_{}_force.pdf".format(gene_name, rank)
    plt.savefig(filename)
    plt.show()
    plt.close()
